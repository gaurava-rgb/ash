select
  count(*) filter (where status <> 'Complete') as open_requests,
  count(*) filter (where assignee = 'Unassigned' and status <> 'Complete') as unassigned,
  count(*) filter (where overdue) as overdue,
  count(*) filter (where status = 'Blocked') as blocked,
  (select count(*) from equipment where active and condition <> 'Available') as equipment_down
from request_overview;
